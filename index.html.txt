function handleImageFile(e) {
    const file = e.target.files && e.target.files[0];
    if (!file) return;
    if (!file.type.startsWith('image/')) { showToast('Это не изображение', 'error'); return; }
    if (file.size > 5 * 1024 * 1024) { showToast('Файл слишком большой (макс. 5 МБ)', 'error'); return; }
    showToast('Обработка фото...', 'info');
    const reader = new FileReader();
    reader.onload = function(evt) {
        const img = new Image();
        img.onload = function() {
            const maxSize = 400;
            let w = img.width, h = img.height;
            if (w > h) { if (w > maxSize) { h = h * maxSize / w; w = maxSize; } }
            else { if (h > maxSize) { w = w * maxSize / h; h = maxSize; } }
            const canvas = document.createElement('canvas');
            canvas.width = Math.round(w); canvas.height = Math.round(h);
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
            const dataUrl = canvas.toDataURL('image/jpeg', 0.7);
            document.getElementById('f-image').value = dataUrl;
            renderImagePreview(dataUrl);
            showToast('Фото загружено');
        };
        img.onerror = function() { showToast('Не удалось прочитать изображение', 'error'); };
        img.src = evt.target.result;
    };
    reader.readAsDataURL(file);
    e.target.value = '';
}

}

const C_KEY = 'vapehab_cart_v2';
function loadCart() {
    try { cart = JSON.parse(localStorage.getItem(C_KEY) || '[]'); } catch(e) { cart = []; }
    updateCartBadge();
    renderCart();
}
function saveCart() {
    localStorage.setItem(C_KEY, JSON.stringify(cart));
    updateCartBadge();
    renderCart();
}
function updateCartBadge() {
    const badge = document.getElementById('cart-badge');
    if (!badge) return;
    const total = cart.reduce(function(s, i) { return s + i.qty; }, 0);
    if (total > 0) { badge.style.display = 'block'; badge.textContent = total; }
    else badge.style.display = 'none';
}

function renderCart() {
    const container = document.getElementById('cart-content');
    if (!container) return;
    if (!cart.length) {
        container.innerHTML = '<div class="cart-empty"><div class="cart-empty-icon">🛒</div><div class="cart-empty-text">Корзина пуста</div></div>';
        return;
    }
    const total = cart.reduce(function(s, i) { return s + i.price * i.qty; }, 0);
    const totalQty = cart.reduce(function(s, i) { return s + i.qty; }, 0);
    const itemsHtml = cart.map(function(item, idx) {
        const img = item.image || 'data:image/svg+xml,%3Csvg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="%23333"%3E%3Cpath d="M21 19V5a2 2 0 0 0-2-2H5a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2z"/%3E%3C/svg%3E';
        return '<div class="cart-item-wrap"><div class="cart-item"><img class="cart-item-img" src="' + img + '" alt=""><div class="cart-item-info"><div><div class="cart-item-name">' + escapeHtml(item.name) + '</div>' + (item.flavor ? '<div class="cart-item-flavor">' + escapeHtml(item.flavor) + '</div>' : '') + '</div><div class="cart-item-bottom"><div class="cart-item-price">' + (item.price * item.qty) + '₽</div><div class="qty-controls"><button class="qty-btn" data-action="qty-down" data-idx="' + idx + '">−</button><span class="qty-value">' + item.qty + '</span><button class="qty-btn" data-action="qty-up" data-idx="' + idx + '">+</button></div></div></div></div><button class="cart-item-remove" data-action="remove-cart" data-idx="' + idx + '"><svg viewBox="0 0 24 24"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg></button></div>';
    }).join('');
    container.innerHTML = itemsHtml + '<div class="cart-summary"><div class="cart-summary-row"><span>Товаров:</span><span>' + totalQty + ' шт.</span></div><div class="cart-summary-row total"><span>Итого:</span><span class="val">' + total + '₽</span></div><button class="cart-checkout-btn" data-action="open-checkout">Оформить заказ</button></div>';
}

function changeQty(idx, delta) {
    if (!cart[idx]) return;
    cart[idx].qty += delta;
    if (cart[idx].qty <= 0) cart.splice(idx, 1);
    saveCart();
}
function removeFromCart(idx) {
    if (!cart[idx]) return;
    cart.splice(idx, 1);
    saveCart();
}

function addToCart(product, flavorName) {
    const key = product.id + (flavorName ? '::' + flavorName : '');
    const existing = cart.find(function(i) { return i.key === key; });
    if (existing) existing.qty += 1;
    else cart.push({ key: key, productId: product.id, name: product.name, flavor: flavorName || null, price: product.price, image: product.image || '', qty: 1 });
    saveCart();
    showToast('Добавлено: ' + product.name + (flavorName ? ' · ' + flavorName : ''));
}

function quickAddToCart(productId) {
    const p = products.find(function(x) { return x.id === productId; });
    if (!p) return;
    addToCart(p, null);
}

function openCheckout() {
    if (!cart.length) { showToast('Корзина пуста', 'error'); return; }
    const firstName = (currentUser && currentUser.first_name) || '';
    const lastName = (currentUser && currentUser.last_name) || '';
    document.getElementById('co-name').value = (firstName + ' ' + lastName).trim();
    document.getElementById('co-phone').value = '';
    document.getElementById('co-address').value = '';
    document.getElementById('co-comment').value = '';
    document.getElementById('checkout-items').innerHTML = cart.map(function(item) {
        return '<div class="checkout-item-row"><div><div>' + escapeHtml(item.name) + ' × ' + item.qty + '</div>' + (item.flavor ? '<div style="font-size:10px;color:#e63946;">' + escapeHtml(item.flavor) + '</div>' : '') + '</div><div>' + (item.price * item.qty) + '₽</div></div>';
    }).join('');
    const total = cart.reduce(function(s, i) { return s + i.price * i.qty; }, 0);
    document.getElementById('checkout-total').textContent = total + '₽';
    selectDelivery('delivery');
    document.getElementById('checkout-modal').classList.add('active');
}
function closeCheckout() { document.getElementById('checkout-modal').classList.remove('active'); }

async function submitOrder() {
    const name = document.getElementById('co-name').value.trim();
    const phone = document.getElementById('co-phone').value.trim();
    const address = document.getElementById('co-address').value.trim();
    const comment = document.getElementById('co-comment').value.trim();
    if (!name) { showToast('Введи имя', 'error'); return; }
    if (!phone) { showToast('Введи телефон', 'error'); return; }
    if (selectedDelivery === 'delivery' && !address) { showToast('Введи адрес', 'error'); return; }
    const total = cart.reduce(function(s, i) { return s + i.price * i.qty; }, 0);
    const order = {
        id: 'o_' + Date.now() + '_' + Math.random().toString(36).slice(2, 6),
        userId: (currentUser && currentUser.id) || 0,
        username: currentUsername || '',
        firstName: (currentUser && currentUser.first_name) || '',
        lastName: (currentUser && currentUser.last_name) || '',
        tgName: name, phone: phone,
        address: selectedDelivery === 'pickup' ? 'Самовывоз' : address,
        deliveryMethod: selectedDelivery, comment: comment,
        items: cart.map(function(i) { return { name: i.name, flavor: i.flavor, price: i.price, qty: i.qty }; }),
        total: total, status: 'new', createdAt: new Date().toISOString()
    };
    orders.unshift(order);
    cart = [];
    saveCart();
    const btn = document.getElementById('co-submit');
    btn.disabled = true; btn.textContent = 'Отправка...';
    await saveDataToCloud();
    btn.disabled = false; btn.textContent = 'Подтвердить заказ';
    closeCheckout();
    updateAdminOrdersBadge();
    showToast('Заказ оформлен!');
    switchTab('orders');
}

function renderAdminTabs() {
    const container = document.getElementById('admin-tabs');
    const tabs = [];
    if (hasPerm('products')) tabs.push({ id: 'add', label: 'Добавить' });
    if (hasPerm('products')) tabs.push({ id: 'list', label: 'Товары' });
    if (hasPerm('orders')) tabs.push({ id: 'orders', label: 'Заказы' + (orders.filter(function(o){return o.status==='new';}).length ? ' 🔴' : '') });
    if (hasPerm('admins')) tabs.push({ id: 'admins', label: 'Админы' });

function adminTab(tab) {
    document.querySelectorAll('.admin-tab').forEach(function(t) { t.classList.toggle('active', t.dataset.tab === tab); });
    document.querySelectorAll('.admin-panel').forEach(function(p) { p.classList.remove('active'); });
    const panel = document.getElementById('panel-' + tab);
    if (panel) panel.classList.add('active');
    if (tab === 'list') renderAdminList();
    if (tab === 'admins') renderAdminsList();
    if (tab === 'orders') renderAdminOrdersList();
    if (tab === 'add') updateFlavorsVisibility();
}

function parseFlavors(text) {
    if (!text) return [];
    return text.split('\n').map(function(l) { return l.trim(); }).filter(function(l) { return l.length; })
        .map(function(line) {
            const parts = line.split('|').map(function(s) { return s.trim(); });
            return { name: parts[0] || '', stock: parseInt(parts[1]) || 0 };
        }).filter(function(f) { return f.name; });
}
function flavorsToString(flavors) {
    if (!flavors || !flavors.length) return '';
    return flavors.map(function(f) { return f.name + ' | ' + f.stock; }).join('\n');
}

async function saveProduct() {
    if (!hasPerm('products')) return;
    const name = document.getElementById('f-name').value.trim();
    const category = document.getElementById('f-category').value;
    const price = parseInt(document.getElementById('f-price').value) || 0;
    const oldPrice = parseInt(document.getElementById('f-oldprice').value) || 0;
    const image = document.getElementById('f-image').value.trim();
    const specs = document.getElementById('f-specs').value.trim();
    const description = document.getElementById('f-description').value.trim();
    const badge = document.getElementById('f-badge').value;
    const editId = document.getElementById('edit-id').value;

    const info = getVariantInfo(category);
    const hasVariants = !!info;
    const flavors = hasVariants ? parseFlavors(document.getElementById('f-flavors').value) : [];
    const manualStock = parseInt(document.getElementById('f-stock').value) || 0;

    if (!name || !price) { showToast('Заполните название и цену', 'error'); return; }
    if (!hasVariants && manualStock <= 0) { showToast('Укажите количество на складе', 'error'); return; }

    const totalStock = hasVariants
        ? (flavors.length ? flavors.reduce(function(s, f) { return s + (f.stock || 0); }, 0) : 0)
        : manualStock;

    const data = {
        name: name, category: category, price: price, oldPrice: oldPrice,
        image: image, specs: specs, flavors: flavors, description: description,
        badge: badge, stock: totalStock
    };

    if (editId) {
        const idx = products.findIndex(function(p) { return p.id === editId; });
        if (idx !== -1) products[idx] = Object.assign({}, products[idx], data);
    } else {
        products.unshift(Object.assign({ id: 'p_' + Date.now() + '_' + Math.random().toString(36).slice(2, 7) }, data));
    }
    const btn = document.getElementById('submit-btn');
    btn.disabled = true; btn.textContent = 'Сохранение...';
    renderProducts();
    renderAdminList();
    clearForm();
    const ok = await saveDataToCloud();
    btn.disabled = false;
    if (ok) showToast(editId ? 'Обновлено' : 'Добавлено');
}

    document.getElementById('f-name').value = '';
    document.getElementById('f-category').value = 'pods';
    document.getElementById('f-price').value = '';
    document.getElementById('f-oldprice').value = '';
    document.getElementById('f-image').value = '';
    document.getElementById('f-specs').value = '';
    document.getElementById('f-flavors').value = '';
    document.getElementById('f-stock').value = '';
    document.getElementById('f-description').value = '';
    document.getElementById('f-badge').value = '';
    document.getElementById('submit-btn').textContent = 'Добавить товар';

    updateFlavorsVisibility();
}

function editProduct(id) {
    if (!hasPerm('products')) return;
    const p = products.find(function(x) { return x.id === id; });
    if (!p) return;
    document.getElementById('edit-id').value = p.id;
    document.getElementById('f-name').value = p.name || '';
    document.getElementById('f-category').value = p.category || 'pods';
    document.getElementById('f-price').value = p.price || '';
    document.getElementById('f-oldprice').value = p.oldPrice || '';
    document.getElementById('f-image').value = p.image || '';
    document.getElementById('f-specs').value = p.specs || '';
    document.getElementById('f-flavors').value = flavorsToString(p.flavors);
    const hasVariants = !!getVariantInfo(p.category);
    document.getElementById('f-stock').value = hasVariants ? '' : (p.stock || '');
    document.getElementById('f-description').value = p.description || '';
    document.getElementById('f-badge').value = p.badge || '';
    document.getElementById('submit-btn').textContent = 'Сохранить изменения';
    renderImagePreview(p.image || '');
    updateFlavorsVisibility();
    adminTab('add');
}

async function deleteProduct(id) {
    if (!hasPerm('products')) return;
    if (!confirm('Удалить товар?')) return;
    products = products.filter(function(p) { return p.id !== id; });
    renderProducts();
    renderAdminList();
    await saveDataToCloud();
}

function renderAdminList() {
    const container = document.getElementById('admin-products-list');
    if (!container) return;
    if (!products.length) { container.innerHTML = '<div class="admin-empty">Товаров пока нет</div>'; return; }
    container.innerHTML = products.map(function(p) {
        const info = getVariantInfo(p.category);
        const variantsInfo = p.flavors && p.flavors.length ? ' · ' + p.flavors.length + ' ' + (info ? info.singular : 'вар.') + '.' : '';
        const stockInfo = !getVariantInfo(p.category) && p.stock ? ' · ' + p.stock + ' шт.' : '';
        return '<div class="admin-product-row"><img class="admin-product-img" src="' + (p.image || '') + '" alt="" onerror="this.style.display=\'none\'"><div class="admin-product-info"><div class="admin-product-name">' + escapeHtml(p.name) + '</div><div class="admin-product-meta">' + (CAT_NAMES[p.category] || p.category) + ' • ' + p.price + '₽' + variantsInfo + stockInfo + '</div></div><div class="admin-product-actions"><button class="admin-action-btn edit" data-action="edit-product" data-id="' + p.id + '"><svg viewBox="0 0 24 24"><path d="M11 4H4a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-7"/><path d="M18.5 2.5a2.121 2.121 0 0 1 3 3L12 15l-4 1 1-4 9.5-9.5z"/></svg></button><button class="admin-action-btn delete" data-action="delete-product" data-id="' + p.id + '"><svg viewBox="0 0 24 24"><polyline points="3 6 5 6 21 6"/><path d="M19 6l-1 14a2 2 0 0 1-2 2H8a2 2 0 0 1-2-2L5 6"/></svg></button></div></div>';
    }).join('');

function togglePerm(perm) {
    const idx = newAdminPerms.indexOf(perm);
    if (idx === -1) newAdminPerms.push(perm);
    else newAdminPerms.splice(idx, 1);
    document.getElementById('cb-' + perm).classList.toggle('checked', newAdminPerms.indexOf(perm) !== -1);
}
function resetPermCheckboxes() {
    newAdminPerms = [];
    ['products', 'orders', 'admins'].forEach(function(p) { document.getElementById('cb-' + p).classList.remove('checked'); });
}

async function addAdmin() {
    if (!hasPerm('admins')) return;
    const input = document.getElementById('new-admin-username');
    let username = input.value.trim().replace(/^@/, '');
    if (!username) { showToast('Введи username', 'error'); return; }
    if (username === OWNER_USERNAME) { showToast('Владелец уже админ', 'error'); return; }
    if (customAdmins.find(function(a) { return a.username === username; })) { showToast('Уже админ', 'error'); return; }
    if (!newAdminPerms.length) { showToast('Выбери хотя бы одно право', 'error'); return; }
    customAdmins.push({ username: username, perms: newAdminPerms.slice() });
    input.value = '';
    resetPermCheckboxes();
    renderAdminsList();
    const ok = await saveDataToCloud();
    if (ok) showToast('@' + username + ' теперь админ');
}

async function removeAdmin(username) {
    if (!hasPerm('admins')) return;
    if (username === OWNER_USERNAME) { showToast('Владельца нельзя удалить', 'error'); return; }
    if (!confirm('Забрать админку у @' + username + '?')) return;
    customAdmins = customAdmins.filter(function(a) { return a.username !== username; });
    renderAdminsList();
    await saveDataToCloud();
}

function editAdminPerms(username) {
    if (!hasPerm('admins')) return;
    if (username === OWNER_USERNAME) { showToast('У владельца все права', 'error'); return; }
    const admin = customAdmins.find(function(a) { return a.username === username; });
    if (!admin) return;
    const currentNums = admin.perms.map(function(p) { return ALL_PERMS.indexOf(p) + 1; }).join(',');
    const newPerms = prompt('Права для @' + username + '\n\n1 — Товары\n2 — Заказы\n3 — Админы\n\nВведи цифры через запятую:', currentNums);
    if (newPerms === null) return;
    const parsed = newPerms.split(',').map(function(s) { return s.trim(); }).filter(function(s) { return s.match(/^[123]$/); }).map(function(n) { return ALL_PERMS[parseInt(n) - 1]; }).filter(Boolean);
    const unique = [];
    parsed.forEach(function(p) { if (unique.indexOf(p) === -1) unique.push(p); });
    admin.perms = unique;
    renderAdminsList();
    saveDataToCloud().then(function(ok) { if (ok) showToast('Права обновлены'); });
}

async function setOrderStatus(id, status) {

function filterByCategory(cat) { currentCategory = cat; renderProducts(); }

function getProductTotalStock(p) {
    if (!p.flavors || !p.flavors.length) return p.stock || 0;
    return p.flavors.reduce(function(s, f) { return s + (f.stock || 0); }, 0);
}

function renderProducts() {
    const grid = document.getElementById('products-grid');
    const emptyState = document.getElementById('empty-state');
    const countEl = document.getElementById('products-count');
    const titleEl = document.getElementById('products-title-text');
    if (!grid) return;
    const q = (document.getElementById('search-input').value || '').toLowerCase().trim();
    let filtered = products.slice();
    if (currentCategory !== 'all') filtered = filtered.filter(function(p) { return p.category === currentCategory; });
    if (q) {
        filtered = filtered.filter(function(p) {
            if ((p.name || '').toLowerCase().indexOf(q) !== -1) return true;
            if ((p.description || '').toLowerCase().indexOf(q) !== -1) return true;
            if (p.flavors && p.flavors.some(function(f) { return (f.name || '').toLowerCase().indexOf(q) !== -1; })) return true;
            return false;
        });
    }
    titleEl.textContent = CAT_NAMES[currentCategory] || 'Каталог';
    countEl.textContent = filtered.length ? filtered.length + ' шт.' : '';
    if (!filtered.length) { grid.innerHTML = ''; emptyState.style.display = 'block'; return; }
    emptyState.style.display = 'none';
    grid.innerHTML = filtered.map(function(p) {
        const badge = p.badge ? '<span class="product-badge ' + p.badge.toLowerCase() + '">' + p.badge + '</span>' : '';
        const oldPrice = p.oldPrice ? '<span class="product-old-price">' + p.oldPrice + '₽</span>' : '';
        const imgSrc = p.image || 'data:image/svg+xml,%3Csvg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="%23333"%3E%3Cpath d="M21 19V5a2 2 0 0 0-2-2H5a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2z"/%3E%3C/svg%3E';
        let variantTags = '';
        if (p.flavors && p.flavors.length) {
            const shown = p.flavors.slice(0, 2);
            variantTags = '<div class="product-flavors">' + shown.map(function(f) { return '<span class="flavor-tag">' + escapeHtml(f.name) + '</span>'; }).join('') + (p.flavors.length > 2 ? '<span class="flavor-more">Ещё ' + (p.flavors.length - 2) + '</span>' : '') + '</div>';
        }
        const specs = p.specs ? '<div class="product-specs">' + escapeHtml(p.specs) + '</div>' : '';
        const totalStock = getProductTotalStock(p);
        const stockHtml = totalStock > 0 ? '<div class="product-stock"><span class="dot"></span>В наличии' + (!p.flavors || !p.flavors.length ? ' · ' + totalStock + ' шт.' : '') + '</div>' : '<div class="product-stock out"><span class="dot"></span>Нет в наличии</div>';
        const info = getVariantInfo(p.category);
        const needVariants = !!info;
        const btnText = needVariants ? (p.category === 'pods' ? 'Выбрать цвет' : 'Выбрать вкус') : 'В корзину';
        const btnAction = needVariants ? 'open-flavor' : 'quick-add';
        return '<div class="product-card"><div class="product-wrap"><img class="product-image" src="' + imgSrc + '" alt="' + escapeHtml(p.name) + '" loading="lazy" onerror="this.src=\'data:image/svg+xml,%3Csvg xmlns=%22http://www.w3.org/2000/svg%22 viewBox=%220 0 24 24%22 fill=%22%23333%22%3E%3Cpath d=%22M21 19V5a2 2 0 0 0-2-2H5a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2z%22/%3E%3C/svg%3E\'">' + badge + '</div><div class="product-info"><div class="product-category">' + (CAT_NAMES[p.category] || p.category) + '</div><div class="product-name">' + escapeHtml(p.name) + '</div>' + variantTags + specs + '<div class="product-price-row"><span class="product-price">' + p.price + '₽</span>' + oldPrice + '</div>' + stockHtml + '<button class="btn-flavor" data-action="' + btnAction + '" data-id="' + p.id + '">' + btnText + '</button></div></div>';
    }).join('');
}

function escapeHtml(s) {
    return String(s || '').replace(/[&<>"']/g, function(c) {
        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
}

/* ===== BOTTOM SHEET ВАРИАНТОВ ===== */
function openFlavorSheet(productId) {
    const p = products.find(function(x) { return x.id === productId; });
    if (!p) return;
    if (!p.flavors || !p.flavors.length) { quickAddToCart(productId); return; }
    currentProduct = p;
    currentFlavorIndex = -1;
    document.getElementById('fs-img').src = p.image || '';
    document.getElementById('fs-cat').textContent = CAT_NAMES[p.category] || p.category;
    document.getElementById('fs-name').textContent = p.name;
    document.getElementById('fs-price').innerHTML = p.price + '₽ <span class="unit">/шт</span>';

    const info = getVariantInfo(p.category) || { word: 'ВАРИАНТ', singular: 'вариант' };
    document.getElementById('fs-header-word').textContent = info.word;

    const availableCount = (p.flavors || []).filter(function(f) { return (f.stock || 0) > 0; }).length;
    const countEl = document.getElementById('fs-count');
    if (availableCount === 0) {
        countEl.textContent = 'Нет в наличии';
        countEl.style.color = '#e63946';
    } else {
        countEl.textContent = 'Доступно: ' + availableCount + ' ' + declineWord(availableCount, p.category);
        countEl.style.color = '';
    }

    renderFlavorList();
    updateSheetCTA();
    document.getElementById('flavor-sheet').classList.add('active');
}

function renderFlavorList() {
    const list = document.getElementById('fs-list');
    const flavors = currentProduct.flavors || [];
    if (!flavors.length) { list.innerHTML = ''; return; }
    list.innerHTML = flavors.map(function(f, i) {
        const inStock = (f.stock || 0) > 0;
        const selected = i === currentFlavorIndex;
        const stockText = inStock ? '<svg viewBox="0 0 24 24"><polygon points="13 2 3 14 12 14 11 22 21 10 12 10 13 2"/></svg> Осталось: ' + f.stock + ' шт.' : '✕ Закончился';
        const stockClass = inStock ? 'avail' : 'out';
        const btnClass = 'flavor-select-btn' + (selected ? ' selected' : '') + (inStock ? '' : ' disabled');
        const btnText = !inStock ? 'Нет' : (selected ? '✓ Выбрано' : '+ Выбрать');
        const dataAction = inStock ? ' data-action="select-flavor" data-idx="' + i + '"' : '';
        return '<div class="flavor-item ' + (selected ? 'selected' : '') + ' ' + (!inStock ? 'out' : '') + '"><div class="flavor-item-info"><div class="flavor-item-name">' + escapeHtml(f.name) + '</div><div class="flavor-item-stock ' + stockClass + '">' + stockText + '</div></div><button class="' + btnClass + '"' + dataAction + '>' + btnText + '</button></div>';
    }).join('');
}

function selectFlavor(index) {
    const f = currentProduct.flavors[index];
    if (!f || (f.stock || 0) <= 0) return;
    if (currentFlavorIndex === index) currentFlavorIndex = -1;
    else currentFlavorIndex = index;
    renderFlavorList();
    updateSheetCTA();
}

function updateSheetCTA() {
    const cta = document.getElementById('fs-cta');
    const info = getVariantInfo(currentProduct.category) || { singular: 'вариант' };
    if (currentFlavorIndex === -1) {
        cta.classList.remove('active');
        cta.textContent = 'Выберите хотя бы 1 ' + info.singular;
    } else {
        const f = currentProduct.flavors[currentFlavorIndex];
        cta.classList.add('active');
        cta.textContent = 'Добавить «' + f.name + '» · ' + currentProduct.price + '₽';
    }
}

function closeFlavorSheet() {
    document.getElementById('flavor-sheet').classList.remove('active');
    currentProduct = null;
    currentFlavorIndex = -1;
}

function addToCartFromSheet() {
    if (currentFlavorIndex === -1 || !currentProduct) return;
    const f = currentProduct.flavors[currentFlavorIndex];
    addToCart(currentProduct, f.name);
    closeFlavorSheet();
}

function switchTab(tab) {
    document.querySelectorAll('.nav-item').forEach(function(el) { el.classList.toggle('active', el.dataset.tab === tab); });
    document.querySelectorAll('.screen').forEach(function(s) { s.classList.remove('active'); });
    const headerTitle = document.querySelector('.header-title-main');
    if (tab === 'catalog') { document.getElementById('catalog-screen').classList.add('active'); headerTitle.innerText = 'VapeHab'; }
    else if (tab === 'cart') { document.getElementById('cart-screen').classList.add('active'); headerTitle.innerText = 'Корзина'; renderCart(); }
    else if (tab === 'orders') { document.getElementById('orders-screen').classList.add('active'); headerTitle.innerText = 'Мои заказы'; renderUserOrders(); }
    else if (tab === 'profile') { document.getElementById('profile-screen').classList.add('active'); headerTitle.innerText = 'Профиль'; }
    else { headerTitle.innerText = tab.charAt(0).toUpperCase() + tab.slice(1); }
    const activeScreen = document.querySelector('.screen.active');
    if (activeScreen) activeScreen.scrollTop = 0;
}

document.addEventListener('click', function(e) {
    const el = e.target.closest('[data-action]');
    if (!el) {
        if (e.target.classList.contains('modal-overlay') && e.target.classList.contains('active')) e.target.classList.remove('active');
        if (e.target.classList.contains('sheet-overlay') && e.target.classList.contains('active')) e.target.classList.remove('active');
        return;
    }
    const action = el.dataset.action;
    switch (action) {
        case 'open-about': openAbout(); break;
        case 'close-about': closeAbout(); break;
        case 'open-manager': openManager(); break;
        case 'open-channel': openChannel(); break;
        case 'open-support': openSupport(); break;
        case 'open-admin': openAdmin(); break;
        case 'close-admin': closeAdmin(); break;
        case 'open-checkout': openCheckout(); break;
        case 'close-checkout': closeCheckout(); break;
        case 'submit-order': submitOrder(); break;
        case 'refresh-data': refreshData(); break;
        case 'switch-tab': switchTab(el.dataset.tab); break;
        case 'filter-category': filterByCategory(el.dataset.cat); break;
        case 'admin-tab': adminTab(el.dataset.tab); break;
        case 'open-flavor': openFlavorSheet(el.dataset.id); break;
        case 'quick-add': quickAddToCart(el.dataset.id); break;
        case 'close-flavor-sheet': closeFlavorSheet(); break;
        case 'select-flavor': selectFlavor(parseInt(el.dataset.idx)); break;
        case 'add-to-cart-from-sheet': addToCartFromSheet(); break;
        case 'qty-up': changeQty(parseInt(el.dataset.idx), 1); break;
        case 'qty-down': changeQty(parseInt(el.dataset.idx), -1); break;
        case 'remove-cart': removeFromCart(parseInt(el.dataset.idx)); break;
        case 'select-delivery': selectDelivery(el.dataset.method); break;
        case 'save-product': saveProduct(); break;
        case 'edit-product': editProduct(el.dataset.id); break;
        case 'delete-product': deleteProduct(el.dataset.id); break;
        case 'toggle-perm': togglePerm(el.dataset.perm); break;

        case 'edit-admin-perms': editAdminPerms(el.dataset.username); break;
        case 'remove-admin': removeAdmin(el.dataset.username); break;
        case 'set-order-status': setOrderStatus(el.dataset.id, el.dataset.status); break;
        case 'upload-image': openImagePicker(); break;
        case 'clear-image': clearImage(); break;
    }
});
